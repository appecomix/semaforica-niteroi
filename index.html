const sheetTrans = ss.getSheetByName("Transferido") || ss.getSheetByName("TRANSFERIDO"); if (!sheetTrans) { return ContentService.createTextOutput(JSON.stringify({ result: "error", error: "Aba Transferido não encontrada." })) .setMimeType(ContentService.MimeType.JSON); } const rows = sheetTrans.getDataRange().getValues(); if (rows.length === 0) { return ContentService.createTextOutput(JSON.stringify({ result: "error", error: "Cabeçalhos não encontrados na aba Transferido." })) .setMimeType(ContentService.MimeType.JSON); } const cabecalhos = rows[0]; let novaLinha = new Array(cabecalhos.length).fill(""); novaLinha[0] = data.data || Utilities.formatDate(new Date(), "GMT-3", "yyyy-MM-dd"); let idxDestino = cabecalhos.findIndex(c => normalizarTexto(c) === "DESTINO" || normalizarTexto(c) === "FILIAL"); const sheetEstoque = ss.getSheetByName("Estoque") || ss.getSheetByName("ESTOQUE"); if (!sheetEstoque) return; // 1. Mapear Entradas let dadosEntradaMap = {}; if (sheetEntrada) { const rowsEntrada = sheetEntrada.getDataRange().getValues(); if (rowsEntrada.length > 0) { let cabEntrada = rowsEntrada[0]; for (let r = 1; r < rowsEntrada.length; r++) { let dataEntrada = rowsEntrada[r][0]; for (let c = 1; c < cabEntrada.length; c++) { let mat = normalizarTexto(cabEntrada[c]); let qtd = Number(rowsEntrada[r][c]) || 0; if (mat && qtd > 0) { if (!dadosEntradaMap[mat]) dadosEntradaMap[mat] = []; dadosEntradaMap[mat].push({ data: dataEntrada, qtd: qtd }); } } } } } // 2. Mapear Dados (Uso em OS) let totalUtilizado = {}; if (sheetDados) { const rowsDados = sheetDados.getDataRange().getValues(); if (rowsDados.length > 0) { let cabDados = rowsDados[0]; for (let r = 1; r < rowsDados.length; r++) { for (let c = 1; c < cabDados.length; c++) { let mat = normalizarTexto(cabDados[c]); let qtd = Number(rowsDados[r][c]) || 0; if (mat && qtd > 0) { totalUtilizado[mat] = (totalUtilizado[mat] || 0) + qtd; } } } } } // 3. Mapear Transferidos let totalTransferido = {}; if (sheetTransferido) { const rowsTrans = sheetTransferido.getDataRange().getValues(); if (rowsTrans.length > 0) { let cabTrans = rowsTrans[0]; for (let r = 1; r < rowsTrans.length; r++) { for (let c = 1; c < cabTrans.length; c++) { let mat = normalizarTexto(cabTrans[c]); let qtd = Number(rowsTrans[r][c]) || 0; if (mat && qtd > 0) { totalTransferido[mat] = (totalTransferido[mat] || 0) + qtd; } } } } } // 4. Utilities.formatDate(d, "GMT-3", "dd/MM/yyyy") : String(val); }; listaEstoque.push({ material: nomeStr, data_ultima_entrada: formatarData(rows[i][1]), qtd_ultima_entrada: rows[i][2] !== "" ? rows[i][2] : "0", qtd_anterior: rows[i][3] !== "" ? rows[i][3] : "0", data_ultima_os: formatarData(rows[i][4]), estoque: rows[i][5] !== "" ? rows[i][5] : "0" }); } } return ContentService.createTextOutput(JSON.stringify({ result: "success", estoque: listaEstoque })) .setMimeType(ContentService.MimeType.JSON); }
